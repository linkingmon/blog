## Approach
To rebuild the balacne tree, I stored the inorder traversal in the vector, and then break all the `left` and `right` connection (this step is important!), and then connecting them again by spliting the nodes into two clusters having similar number of nodes.

## Code
```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    TreeNode* balanceBST(TreeNode* root) {
        vector<TreeNode*> inorder;
        traverse(inorder, root);
        int n = inorder.size();
        for(TreeNode* node: inorder) node->left = node->right = nullptr;
        return rebuild(inorder, 0, n-1);
    }

    void traverse(vector<TreeNode*>& inorder, TreeNode* root) {
        if(root->left) traverse(inorder, root->left);
        inorder.push_back(root);
        if(root->right) traverse(inorder, root->right);
    }

    TreeNode* rebuild(const vector<TreeNode*>& inorder, int start, int end) {
        if(start == end) return inorder[start];
        if(end < start) return nullptr;
        int m = start + (end - start) / 2;
        TreeNode* curNode = inorder[m];
        curNode->left = rebuild(inorder, start, m-1);
        curNode->right = rebuild(inorder, m+1, end);
        return curNode;
    }
};
```
