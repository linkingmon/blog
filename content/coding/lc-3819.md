## Approach
We record the non-negative ones' idx in a separate array and rotate it, and then traverse the original array and mofitied the non-negaitve number based on the rotatete idx array.

## Code
```c++
class Solution {
public:
    vector<int> rotateElements(vector<int>& nums, int k) {
        int n = nums.size();
        vector<int> idx;
        for(int i = 0 ; i < n ; ++i) {
            if(nums[i] >= 0) idx.push_back(i);
        }
        int m = idx.size();
        vector<int> idx2(m);
        for(int i = 0 ; i < m ; ++i) {
            idx2[i] = idx[(i+k)%m];
        }
        int cnt = 0;
        vector<int>ans = nums;
        for(int i = 0 ; i < n ; ++i) {
            if(ans[i] >= 0) {
                ans[i] = nums[idx2[cnt]];
                ++cnt;
            }
        }
        return ans;
    }
};
```
