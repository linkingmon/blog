check the currnet node value, and then check the status for left and right child with the function itself recursively.

## Code
```c++
class Solution {
public:
    bool isSameTree(TreeNode* p, TreeNode* q) {
        if (p == nullptr && q == nullptr) return true;
        else if (p == nullptr || q == nullptr) return false;

        if (p->val != q->val) return false;

        if ((p->left == nullptr) ^ (q->left == nullptr)) return false;   
        else if (!isSameTree(p->left, q->left)) return false;
        
        if((p->right == nullptr) ^ (q->right == nullptr)) return false;   
        else if (!isSameTree(p->right, q->right)) return false;
        
        return true;
    }
};
```
