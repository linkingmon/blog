After the contest ended, I saw other people’s solutions and realized that because the problem restricts the values and the number of elements in nums, the maximum product is simply the product of numbers from 1 to 9. So once the multiplication exceeds that bound, we can stop checking further.

This taught me that when I see potentially huge numbers, I should check whether there’s actually a built-in upper bound.

## Code
```c++
class Solution {
public:
    int maxLength(vector<int>& nums) {
        int n = nums.size();
        int maxLen = 1;

        for(int i = 0 ; i < n ; ++i) {
            int l = nums[i];
            int g = nums[i];
            long long prod = nums[i];
            for(int j = i+1 ; j < n ; ++j) {
                l = lcm(l, nums[j]);
                g = gcd(g, nums[j]);
                prod *= nums[j];
                if(prod > 10'000'000'000L) break;
                if(l*g == prod) maxLen = max(maxLen, j-i+1);
            }
        }

        return maxLen;
    }
};
```
