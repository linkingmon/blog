Count hte frequency, and sort by the freq in descending order, then return the first `k` number.

## Code
```c++
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        unordered_map<int, int> freq;
        for(const auto& x: nums) ++freq[x];

        vector<pair<int,int>> v;
        for(const auto& [x, cnt]: freq) v.push_back({cnt, x});
        sort(v.begin(), v.end(), greater<pair<int,int>>());
        
        vector<int> ans;
        ans.reserve(k);
        for(int i = 0 ; i < k ; ++i) ans.push_back(v[i].second);

        return ans;
    }
};
```
