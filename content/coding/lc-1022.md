traverse the tree and maintain current number, double the value when approaching next level, and sum the value up when reach leaf.

## Code
```c++
class Solution {
public:
    int sumRootToLeaf(TreeNode* root) {
        int sum = 0;
        traverse(0, root, sum);
        return sum;
    }

    void traverse(int n, TreeNode* root, int& sum) {
        n = (n << 1) + root->val;
        if(root->left) traverse(n, root->left, sum);
        if(root->right) traverse(n, root->right, sum);
        if(root->left == nullptr && root->right == nullptr) sum += n;
    }
};
```
