In my first attempt, I didn’t notice the possibility of an infinite loop, so I needed to maintain a visited set.

At the beginning I wasn’t very sure how to estimate the complexity of this problem. It feels like it should be `O(n)`. My reasoning was that in the worst case every element is 1, and we start searching from the end all the way to the beginning. As long as we ensure that we never visit the same index twice, the complexity stays `O(n)`.

## Code
```c++
class Solution {
public:
    bool canReach(vector<int>& arr, int start) {
        vector<bool> visited(arr.size(), false);
        return traverse(arr, start, visited);
    }
    bool traverse(vector<int>& arr, int start, vector<bool>& visited) {
        if(visited[start]) {
            return false;
        }
        visited[start] = true;
        if(arr[start] == 0) {
            return true;
        }
        if(start + arr[start] < arr.size()) {
            if(traverse(arr, start+arr[start], visited)) {
                return true;
            }
        }
        if(start - arr[start] >= 0) {
            if(traverse(arr, start-arr[start], visited)) {
                return true;
            }
        }
        return false;
    }
};
```
