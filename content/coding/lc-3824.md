## Approach
This is a typical search on answer problme, we try to see if the answer is valid, this can be done in simple math by `O(N)`, and the binary search on `lgN`.

## Code
```c++
class Solution {
public:
    int minimumK(vector<int>& nums) {
        bool isAllNegative = true;
        int n = nums.size();
        for(int i = 0 ; i < n ; ++i) {
            if(nums[i] > 0) {
                isAllNegative = false;
                break;
            }
        }
        if(isAllNegative) return 0;
        int l = 1;
        int r = 100000;
        while(l < r) {
            int m = l + (r - l) / 2;
            if(isValid(nums, m)) {
                r = m;
            } else {
                l = m + 1;
            }
        }
        return l;
    }

    bool isValid(const vector<int>& v, int k) {
        int n = v.size();
        int cnt = 0;
        for(int i = 0 ; i < n ; ++i) {
            if(v[i] <= 0) continue;
            cnt += v[i] / k;
            if(v[i] % k != 0) ++cnt;
        }
        return (long long)cnt <= (long long)k*k;
    }
};
```
