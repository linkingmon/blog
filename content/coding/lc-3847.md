Check if the current number is odd, and the idx is mod 5 by 6, then update `score1` and `score2`

## Code
```c++
class Solution {
public:
    int scoreDifference(vector<int>& nums) {
        int score1 = 0;
        int score2 = 0;
        bool active1 = true;

        int n = nums.size();
        for(int i = 0 ; i < n ; ++i) {
            if((nums[i] % 2 == 1) ^ (i % 6 == 5)) active1 = !active1;
            if(active1) score1 += nums[i];
            else score2 += nums[i];
        }

        return score1 - score2;
    }
};
```
